  - [-1, 1, Conv,  [512, 3, 2]]          # 5-P4/16

  - [-1, 1, Conv,  [1024, 3, 2]]         # 7-P5/32
  - [-1, 3, A2C2f, [1024, True, 1]]      # 8

  - [[-1, 4], 1, Concat, [1]]

  - [-1, 2, C3k2, [1024, True]]          # 23 (P5)

  - [[17, 20, 23], 1, Detect, [nc]]      # 3 heads only — no anchor dilution
